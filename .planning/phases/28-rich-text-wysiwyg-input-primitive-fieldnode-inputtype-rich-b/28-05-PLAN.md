    private richTextToolbar(
      n: RichTextToolbarNode,
      parent: HTMLElement,
      _on: (a: ActionEvent) => void
    ): void {
      const el = document.createElement("div");
      el.className = "vms-rich-text-toolbar";
      if (n.size) el.classList.add(`vms-rich-text-toolbar--size-${n.size}`);
      if (n.tone) el.classList.add(`vms-rich-text-toolbar--tone-${n.tone}`);
      if (n.state) el.classList.add(`vms-rich-text-toolbar--state-${n.state}`);

      for (const tool of n.tools) {
        const btn = document.createElement("button");
        btn.type = "button";
        btn.className = `vms-rich-text-toolbar__tool vms-rich-text-toolbar__tool--${tool}`;
        btn.dataset.tool = tool;
        // Label per D-08 floor (short glyph or letter, matching the spike reference)
        btn.textContent = this.richTextToolLabel(tool);
        btn.setAttribute("aria-label", this.richTextToolAriaLabel(tool));
        btn.addEventListener("click", (ev) => {
          ev.preventDefault();
          this.richTextToolbarInvoke(el, tool);
        });
        el.appendChild(btn);
      }
      parent.appendChild(el);
    }

    /** Resolve the ancestor RichTextField's editor and invoke the TipTap
     *  chain command. Walks up from the toolbar's parent to find a
     *  .vms-rich-text-field container; then queries editorInstances by the
     *  data-editor-key attribute that richTextField() writes onto its
     *  wrapper. Standalone case: no ancestor found → console.warn + no-op. */
    private richTextToolbarInvoke(toolbarEl: HTMLElement, tool: RichTextTool): void {
      const fieldEl = toolbarEl.closest(".vms-rich-text-field") as HTMLElement | null;
      if (!fieldEl) {
        console.warn(
          "[ViewModelShell]",
          "RichTextToolbarNode rendered standalone (no ancestor RichTextFieldNode); toolbar clicks are inert.",
        );
        return;
      }
      const key = fieldEl.dataset.editorKey;
      if (!key) return;
      const entry = this.editorInstances.get(key);
      if (!entry?.editor) return;
      const ed = entry.editor;
      switch (tool) {
        case "bold": ed.chain().focus().toggleBold().run(); break;
        case "italic": ed.chain().focus().toggleItalic().run(); break;
        case "link": {
          const url = window.prompt("URL:") ?? "";
          if (url) ed.chain().focus().setLink({ href: url }).run();
          break;
        }
        case "bullet-list": ed.chain().focus().toggleBulletList().run(); break;
        case "ordered-list": ed.chain().focus().toggleOrderedList().run(); break;
        case "heading-1": ed.chain().focus().toggleHeading({ level: 1 }).run(); break;
        case "heading-2": ed.chain().focus().toggleHeading({ level: 2 }).run(); break;
        case "heading-3": ed.chain().focus().toggleHeading({ level: 3 }).run(); break;
        case "inline-code": ed.chain().focus().toggleCode().run(); break;
        case "code-block": ed.chain().focus().toggleCodeBlock().run(); break;
        case "blockquote": ed.chain().focus().toggleBlockquote().run(); break;
      }
    }
