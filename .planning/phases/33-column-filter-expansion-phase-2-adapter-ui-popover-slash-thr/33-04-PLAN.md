// Fake StateAccess (BrowserAdapter accepts optional stateAccess param in render()):
// NOTE: Read the actual BrowserAdapter render() signature to confirm.
// StateAccess is: { read(path: string): unknown; write(path: string, value: unknown): void }
function makeStateAccess(state: Record<string, unknown>) {
  return {
    read: (path: string) => path.split(".").reduce((o, k) => (o as Record<string, unknown>)?.[k], state),
    write: (path: string, value: unknown) => {
      const parts = path.split(".");
      let o = state;
      for (let i = 0; i < parts.length - 1; i++) {
        if (!(parts[i] in o)) o[parts[i]] = {};
        o = o[parts[i]] as Record<string, unknown>;
      }
      o[parts[parts.length - 1]] = value;
    },
  };
}

// Building test TableNode with filterDescriptorBinds:
function makeFilterTable(colKey: string, kind: FilterSpec["kind"], options?: string[]): TableNode {
  return {
    type: "table",
    columns: [
      { key: colKey, label: "Col", filter: { kind, options } },
    ],
    rows: [],
    filterDescriptorBinds: { [colKey]: `fd.${colKey}` },
  };
}

// Fake getBoundingClientRect for positioning tests:
function fakeRect(el: Element, rect: Partial<DOMRect>) {
  vi.spyOn(el, "getBoundingClientRect").mockReturnValue({
    top: 0, left: 0, bottom: 0, right: 0, width: 0, height: 0,
    x: 0, y: 0, toJSON: () => ({}),
    ...rect,
  });
}
